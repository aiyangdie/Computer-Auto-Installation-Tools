def setup_logging(log_dir="install_logs"):
    """配置日志系统"""
    if not os.path.exists(log_dir):
        os.makedirs(log_dir)
    
    log_file = os.path.join(log_dir, f"install_log_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt")
    logging.basicConfig(
        level=logging.INFO,
        format='%(asctime)s - %(levelname)s - %(message)s',
        handlers=[
            logging.FileHandler(log_file),
            logging.StreamHandler()
        ]
    )
    return log_file

#### 文件查找函数

```python
def find_exe_files(root_dir, include_subfolders=False, name_filter=None, size_min=None, size_max=None):
    """查找指定目录中的所有exe文件，可应用过滤条件"""
    exe_folder = os.path.join(root_dir, "exe")
    if not os.path.exists(exe_folder):
        logging.error(f"exe文件夹不存在: {exe_folder}")
        return []
    
    exe_files = []
    
    # 定义搜索函数，支持递归
    def search_exes(folder):
        for item in os.listdir(folder):
            full_path = os.path.join(folder, item)
            
            # 如果是文件夹且需要包含子文件夹
            if os.path.isdir(full_path) and include_subfolders:
                search_exes(full_path)
                continue
                
            # 非exe文件跳过
            if not item.lower().endswith(".exe"):
                continue
                
            # 应用名称过滤
            if name_filter and not re.search(name_filter, item, re.IGNORECASE):
                continue
                
            # 应用大小过滤
            if size_min is not None or size_max is not None:
                file_size_mb = os.path.getsize(full_path) / (1024 * 1024)  # 转换为MB
                
                if size_min is not None and file_size_mb < size_min:
                    continue
                    
                if size_max is not None and file_size_mb > size_max:
                    continue
            
            # 通过所有过滤条件，添加到列表
            exe_files.append(full_path)
    
    # 开始搜索
    search_exes(exe_folder)
    return exe_files
```

#### 安装执行函数

```python
def install_exe(exe_path, timeout=300, use_defaults=True):
    """尝试安装exe文件，使用常见的静默安装参数"""
    file_name = os.path.basename(exe_path)
    logging.info(f"开始安装: {file_name}")
    
    # 常见的静默安装参数列表，按顺序尝试
    silent_args = [
        "/S",                  # 许多安装程序
        "/silent",             # 一些安装程序
        "/quiet",              # 微软安装程序
        "/verysilent",         # Inno Setup
        "/passive",            # MSI安装程序
        "--silent",            # 一些安装程序
        "-s",                  # 一些安装程序
        "-q",                  # 一些安装程序
        "/q",                  # 微软安装程序
        "/AUTO"                # 一些安装程序
    ]
    
    # 对于某些特定程序的特殊处理
    if "vcredist" in file_name.lower() or "vc_redist" in file_name.lower():
        silent_args = ["/quiet", "/norestart"]
    elif "microsoft" in file_name.lower() and "edge" in file_name.lower():
        silent_args = ["/silent", "/install"]
    elif "git" in file_name.lower():
        silent_args = ["/VERYSILENT", "/NORESTART", "/NOCANCEL", "/SP-"]
    elif "python" in file_name.lower():
        silent_args = ["/quiet", "InstallAllUsers=1", "PrependPath=1", "Include_test=0"]
    elif "nvidia" in file_name.lower():
        silent_args = ["-s", "-noreboot"]
    # ... 更多特殊程序处理 ...
    
    success = False
    
    # 如果需要使用默认安装（不使用静默参数）
    if not use_defaults:
        try:
            logging.info(f"以交互方式运行: {file_name}")
            subprocess.Popen([exe_path], shell=True)
            # 因为交互式安装需要用户操作，所以不等待其完成
            logging.info(f"已启动交互式安装: {file_name}，请手动完成安装")
            return True
        except Exception as e:
            logging.error(f"启动交互式安装 {file_name} 失败: {str(e)}")
            return False
    
    # 首先尝试直接运行，不加参数（有些程序不需要参数）
    try:
        logging.info(f"尝试直接运行: {file_name}")
        result = subprocess.run([exe_path], 
                               shell=True, 
                               capture_output=True, 
                               text=True, 
                               timeout=timeout)
        if result.returncode == 0:
            logging.info(f"成功安装: {file_name} (无参数)")
            return True
    except subprocess.TimeoutExpired:
        logging.warning(f"{file_name} 安装超时，尝试使用静默参数")
    except Exception as e:
        logging.warning(f"直接运行 {file_name} 失败: {str(e)}")
    
    # 尝试每种静默安装参数
    for args in silent_args:
        try:
            cmd = [exe_path]
            # 将参数拆分成列表添加
            for arg in args.split():
                cmd.append(arg)
                
            logging.info(f"尝试安装 {file_name} 使用参数: {args}")
            result = subprocess.run(cmd, 
                                   shell=True, 
                                   capture_output=True, 
                                   text=True, 
                                   timeout=timeout)
            
            if result.returncode == 0:
                logging.info(f"成功安装: {file_name} 使用参数: {args}")
                success = True
                break
            else:
                logging.warning(f"参数 {args} 安装失败: {result.stderr}")
        except subprocess.TimeoutExpired:
            logging.warning(f"{file_name} 使用参数 {args} 安装超时")
        except Exception as e:
            logging.warning(f"使用参数 {args} 安装 {file_name} 时出错: {str(e)}")
    
    if not success:
        logging.error(f"所有尝试都失败，无法安装: {file_name}")
    
    return success
```

这个函数是核心安装逻辑，它尝试使用各种常见的静默安装参数安装exe文件。对于特定的程序（如Python, Git等），它会使用特定的安装参数。如果静默安装失败，它会尝试其他参数，直到成功或尝试完所有参数。

def cli_mode():
    """命令行界面模式"""
    parser = argparse.ArgumentParser(description="自动安装exe文件")

    parser.add_argument("-f", "--filter", help="文件名过滤，支持正则表达式")
    parser.add_argument("--min-size", help="最小文件大小(MB)", type=float)
    parser.add_argument("--max-size", help="最大文件大小(MB)", type=float)
    parser.add_argument("-t", "--timeout", help="安装超时时间(秒)", type=int, default=300)
    parser.add_argument("-i", "--interactive", help="使用交互模式安装，而非静默安装", action="store_true")
    
    args = parser.parse_args()
    
    # 设置日志
    log_file = setup_logging()
    
    # 查找exe文件
    exe_files = find_exe_files(
        args.directory, 
        args.subfolders, 
        args.filter, 
        args.min_size, 
        args.max_size
    )
    
    if not exe_files:
        logging.error("未找到符合条件的exe文件")
        return
    
    logging.info(f"找到 {len(exe_files)} 个exe文件:")
    for i, exe_file in enumerate(exe_files, 1):
        file_size_mb = os.path.getsize(exe_file) / (1024 * 1024)
        logging.info(f"{i}. {os.path.basename(exe_file)} ({file_size_mb:.2f} MB)")
    
    # 安装文件
    success_count = 0
    for exe_file in exe_files:
        if install_exe(exe_file, args.timeout, not args.interactive):
            success_count += 1
    
    logging.info(f"安装完成。成功: {success_count}/{len(exe_files)}")
    logging.info(f"详细日志保存在: {log_file}")

def browse_directory(self):
    """浏览并选择目录"""
    directory = filedialog.askdirectory(initialdir=self.root_dir)
    if directory:
        self.root_dir = directory
        self.dir_entry.delete(0, tk.END)
        self.dir_entry.insert(0, directory)
        self.refresh_exe_list()

def refresh_exe_list(self):
    """刷新exe文件列表"""
    # 清空现有列表
    for item in self.tree.get_children():
        self.tree.delete(item)
    
    # 获取目录
    directory = self.dir_entry.get()
    if not directory:
        messagebox.showerror("错误", "请选择有效的目录")
        return
    
    # 获取过滤条件
    name_filter = self.name_filter.get() if self.name_filter.get() else None
    min_size = self.min_size.get() if self.min_size.get() > 0 else None
    max_size = self.max_size.get() if self.max_size.get() > 0 else None
    
    # 查找exe文件
    exe_files = find_exe_files(
        directory, 
        self.include_subfolders.get(), 
        name_filter, 
        min_size, 
        max_size
    )
    
    # 添加到列表
    for exe_file in exe_files:
        file_size_mb = os.path.getsize(exe_file) / (1024 * 1024)
        self.tree.insert("", tk.END, values=(
            os.path.basename(exe_file),
            f"{file_size_mb:.2f}",
            "待安装"
        ), tags=("pending",))
    
    # 设置标签颜色
    self.tree.tag_configure("pending", background="#f0f0f0")
    self.tree.tag_configure("success", background="#90ee90")  # 浅绿色
    self.tree.tag_configure("failure", background="#ffcccb")  # 浅红色
    
    # 显示结果数量
    if not exe_files:

#### 安装功能方法

```python
def install_selected(self):
    """安装选中的文件"""
    selected_items = self.tree.selection()
    if not selected_items:
        messagebox.showwarning("警告", "请先选择要安装的文件")
        return
    
    # 设置日志
    log_file = setup_logging()
    messagebox.showinfo("信息", f"安装日志将保存在: {log_file}")
    
    # 创建安装线程
    install_thread = threading.Thread(
        target=self.install_thread,
        args=(selected_items,)
    )
    install_thread.daemon = True
    install_thread.start()

def install_thread(self, selected_items):
    """安装线程，避免界面冻结"""
    directory = self.dir_entry.get()
    timeout = self.timeout.get()
    use_interactive = self.use_interactive.get()
    
    total = len(selected_items)
    success_count = 0
    
    for i, item in enumerate(selected_items, 1):
        # 获取文件信息
        values = self.tree.item(item, "values")
        filename = values[0]
        
        # 查找完整路径
        exe_path = None
        for root, _, files in os.walk(os.path.join(directory, "exe")):
            if filename in files:
                exe_path = os.path.join(root, filename)
                break
        
        if not exe_path:
            logging.error(f"找不到文件: {filename}")
            self.root.after(0, lambda item=item: self.tree.item(
                item, tags=("failure",), values=(filename, values[1], "失败")
            ))
            continue
        
        # 更新状态为"安装中"
        self.root.after(0, lambda item=item: self.tree.item(
            item, values=(filename, values[1], "安装中...")
        ))
        
        # 安装文件
        success = install_exe(exe_path, timeout, not use_interactive)
        
        # 更新状态
        if success:
            success_count += 1
            self.root.after(0, lambda item=item: self.tree.item(
                item, tags=("success",), values=(filename, values[1], "成功")
            ))
        else:
            self.root.after(0, lambda item=item: self.tree.item(
                item, tags=("failure",), values=(filename, values[1], "失败")
            ))
    
    # 显示完成消息
    self.root.after(0, lambda: messagebox.showinfo(
        "完成", f"安装完成。成功: {success_count}/{total}"
    ))
```

这些方法实现了安装功能，包括创建安装线程（避免界面冻结）和更新安装状态。

## 简化版模块 (auto_install.py)

简化版模块提供了基本的自动安装功能，没有图形界面，也没有复杂的过滤条件，适合简单场景。

```python
import os
import subprocess
import logging
import sys
import time

# 设置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler("auto_install.log"),
        logging.StreamHandler()
    ]
)

def install_exe(exe_path):
    """安装exe文件，尝试常见的静默安装参数"""
    file_name = os.path.basename(exe_path)
    logging.info(f"开始安装: {file_name}")
    
    # 尝试静默参数列表
    silent_args = ["/S", "/silent", "/quiet", "/verysilent", "/passive"]
    
    # 先尝试直接运行
    try:
        result = subprocess.run([exe_path], shell=True, timeout=300)
        if result.returncode == 0:
            logging.info(f"成功安装: {file_name}")
            return True
    except Exception as e:
        logging.warning(f"直接运行失败: {str(e)}")
    
    # 尝试各种静默参数
    for arg in silent_args:
        try:
            logging.info(f"尝试参数: {arg}")
            result = subprocess.run([exe_path, arg], shell=True, timeout=300)
            if result.returncode == 0:
                logging.info(f"成功安装: {file_name} 使用参数: {arg}")
                return True
        except Exception as e:
            logging.warning(f"参数 {arg} 失败: {str(e)}")
    
    logging.error(f"所有尝试都失败: {file_name}")
    return False

def main():
    """主函数"""
    # 检查exe目录
    exe_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "exe")
    if not os.path.exists(exe_dir):
        logging.error(f"exe目录不存在: {exe_dir}")
        return
    
    # 查找所有exe文件
    exe_files = [os.path.join(exe_dir, f) for f in os.listdir(exe_dir) if f.lower().endswith(".exe")]
    if not exe_files:
        logging.error("未找到exe文件")
        return
    
    logging.info(f"找到 {len(exe_files)} 个exe文件")
    
    # 安装文件
    success_count = 0
    for exe_file in exe_files:
        if install_exe(exe_file):
            success_count += 1
    
    logging.info(f"安装完成。成功: {success_count}/{len(exe_files)}")

if __name__ == "__main__":
    main()
```

## 代码设计特点

1. **双模式设计**：同时支持命令行和图形界面两种操作模式
2. **模块化结构**：将功能分解为独立的模块和函数
3. **智能安装**：自动尝试多种安装参数，适配不同类型的安装包
4. **错误处理**：包含完善的错误处理和日志记录
5. **多线程处理**：使用线程避免界面冻结
6. **用户友好**：提供直观的界面和清晰的反馈
7. **灵活配置**：支持多种过滤条件和安装选项
